{{/* Author profile box — local override of the HugoBlox (blox-bootstrap v5) partial. Adds: `profile: ` in a page's front matter shows that author's profile card at the bottom of the page, without listing them under `authors:`. A list also works: `profile: [anna-lou-abatayo, andries-richter]`. `profile: false` still hides the box, as in the original theme. */}} {{/* Don't show author box on normal pages or if author box disabled. */}} {{ if ne .Type "page" | and (not (or (eq site.Params.profile false) (eq .Params.profile false))) }} {{ $profile := .Params.profile }} {{ $names := slice }} {{ if reflect.IsSlice $profile }}{{ $names = $profile }}{{ else if and $profile (eq (printf "%T" $profile) "string") }}{{ $names = slice $profile }}{{ end }} {{ if $names }} {{/* Show the profile(s) named in `profile:` — one slug, or a list of slugs. */}} {{ range $names }} {{ with site.GetPage (printf "/authors/%s" (urlize .)) }} {{ partial "page_author_card" (dict "author_page" .) }} {{ else }} {{ warnf "page_author: no author profile found for profile %q on %s" . $.File.Path }} {{ end }} {{ end }} {{/* Display superuser if superuser exists and page authors are not explicitly specified. */}} {{ else if and (not .Params.authors) (.Scratch.Get "superuser_username") }} {{ $author_page := site.GetPage (printf "/%s/%s" "authors" (.Scratch.Get "superuser_username")) }} {{ partial "page_author_card" (dict "author_page" $author_page) }} {{/* Otherwise, display each listed author that has a profile. */}} {{ else if .Params.authors }} {{ $taxonomy := "authors" }} {{ range $author_obj := (.GetTerms $taxonomy) }} {{ partial "page_author_card" (dict "author_page" $author_obj.Page) }} {{ end }} {{ end }} {{end}}{{/* Show profile block */}}
